  <script>
    /* ========== PAGE SYSTEM ========== */
    const screens = [...document.querySelectorAll('.screen')];
    let current = 0;
    let isFlipping = false;

    function updateProgress() {
      const percentage = (current / (screens.length - 1)) * 100;
      document.getElementById('bar').style.width = percentage + '%';
    }

    function go(nextPage) {
      if (isFlipping) return;
      if (nextPage < 0 || nextPage >= screens.length || nextPage === current) return;

      const oldPage = screens[current];
      const newPage = screens[nextPage];
      const goingForward = nextPage > current;
      isFlipping = true;

      oldPage.querySelectorAll('video').forEach(v => v.pause());

      oldPage.classList.add(goingForward ? 'flip-out' : 'flip-back');

      setTimeout(() => {
        oldPage.classList.remove('active', 'flip-out', 'flip-back');
        current = nextPage;
        newPage.scrollTop = 0;
        if (!goingForward) newPage.classList.add('from-back');
        newPage.classList.add('active');
        updateProgress();
        setTimeout(() => {
          newPage.classList.remove('from-back');
          isFlipping = false;
        }, 550);
      }, 350);
    }

    /* ========== QUESTION ANSWERS ========== */
    function answer(btn, correct, msg) {
      const card = btn.closest('.card');
      const feedback = card.querySelector('.feedback');
      const choices = card.querySelectorAll('.choice');
      const screen = btn.closest('.screen');
      const next = screen.querySelector('.next');

      if (!correct) {
        btn.classList.add('wrong');
        feedback.className = 'feedback bad';
        feedback.innerHTML = `<div class="wrong-popup">❌ ${msg}</div>`;
        return;
      }

      btn.classList.add('correct');
      choices.forEach(choice => choice.disabled = true);
      feedback.className = 'feedback good';
      feedback.innerHTML = `<div class="correct-popup">✓ ${msg}</div>`;
      if (next) next.classList.remove('hidden');
    }

    /* ========== NEXT BUTTONS ========== */
    document.querySelectorAll('.next').forEach(btn => {
      btn.addEventListener('click', () => go(current + 1));
    });

    /* ========== FINAL ========== */
    function finale() { go(15); }

    updateProgress();
  </script>
