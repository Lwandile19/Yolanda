<script>


/* =========================================
   PAGE SYSTEM
========================================= */

const screens =
    [...document.querySelectorAll('.screen')];

let current = 0;

let isFlipping = false;


/* =========================================
   PROGRESS BAR
========================================= */

function updateProgress() {

    const percentage =
        (current / (screens.length - 1)) * 100;

    document.getElementById('bar')
        .style.width =
        percentage + '%';
}


/* =========================================
   PAGE FLIP
========================================= */

function go(nextPage) {

    if (isFlipping) {
        return;
    }

    if (
        nextPage < 0 ||
        nextPage >= screens.length ||
        nextPage === current
    ) {
        return;
    }

    const oldPage = screens[current];
    const newPage = screens[nextPage];

    const goingForward = nextPage > current;

    isFlipping = true;

    /*
        Stop any videos on the page we're leaving.
    */
    oldPage
        .querySelectorAll('video')
        .forEach(video => video.pause());

    /*
        Flip the current page in the correct direction.
    */
    oldPage.classList.add(
        goingForward ? 'flip-out' : 'flip-back'
    );

    setTimeout(() => {

        oldPage.classList.remove(
            'active',
            'flip-out',
            'flip-back'
        );

        current = nextPage;

        newPage.scrollTop = 0;

        /*
            When going backwards, the new page
            enters from the left.
        */
        if (!goingForward) {
            newPage.classList.add('from-back');
        }

        newPage.classList.add('active');

        updateProgress();

        setTimeout(() => {

            newPage.classList.remove('from-back');

            isFlipping = false;

        }, 550);

    }, 350);
}


/* =========================================
   QUESTION ANSWERS
========================================= */

function answer(btn, correct, msg) {

    const card =
        btn.closest('.card');

    const feedback =
        card.querySelector('.feedback');

    const choices =
        card.querySelectorAll('.choice');

    const next =
        card.parentElement
            .parentElement
            .querySelector('.next');


    /*
        WRONG ANSWER
    */

    if (!correct) {

        /*
            Disable ONLY the wrong
            answer.

            The player can still
            try the other answers.
        */

        btn.disabled = true;

        btn.classList.add('wrong');


        feedback.className =
            'feedback bad';


        feedback.innerHTML =
            `
            <div class="wrong-popup">
                ❌ ${msg}
            </div>
            `;


        /*
            DO NOT reveal NEXT.
        */

        return;
    }


    /*
        CORRECT ANSWER
    */

    btn.classList.add('correct');


    /*
        Lock all choices.
    */

    choices.forEach(choice => {

        choice.disabled = true;

    });


    feedback.className =
        'feedback good';


    feedback.innerHTML =
        `
        <div class="correct-popup">
            ✓ ${msg}
        </div>
        `;


    /*
        NEXT now appears.
    */

    if (next) {

        next.classList.remove(
            'hidden'
        );

    }

}


/* =========================================
   QUESTION NEXT BUTTONS
========================================= */

document
    .querySelectorAll('.next')
    .forEach(button => {

        button.addEventListener(
            'click',
            () => {

                go(current + 1);

            }
        );

    });


/* =========================================
   FINAL ANSWER
========================================= */

function finale() {

    go(15);

}


/* =========================================
   INITIAL PROGRESS
========================================= */

updateProgress();


</script>
